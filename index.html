============================================================ # SISTEMA DE LUTA # ============================================================ def simulate_fight(player: Fighter, opponent_overall: int, is_title: bool = False) -> dict: """Simula uma luta usando taxas reais + atributos do jogador.""" rates = FINISH_RATES[player.peso] p_overall = player.overall # Weight cut ruim aumenta chance de perda for inj in player.active_injuries(): if inj.tipo == "lesao_weight_cut": win_chance *= 0.85 won = random.random() < win_chance # Método de finalização r = random.random() if r < rates["ko"]: method = "KO/TKO" round_num = random.choices([1, 2, 3, 4, 5], weights=[45, 25, 15, 10, 5])[0] elif r < rates["ko"] + rates["sub"]: method = "Submission" round_num = random.choices([1, 2, 3, 4, 5], weights=[40, 30, 20, 7, 3])[0] else: method = "Decision" round_num = 5 if is_title else 3 # Dano e lesões injuries = generate_injuries(player, won, method, is_title) # Wear & Tear wear_gain = 1.5 if method in ["KO/TKO", "Submission"] else 0.8 if not won: wear_gain *= 1.6 if method == "KO/TKO" and not won: wear_gain += 2.5 # KO sofrido dói muito return { "won": won, "method": method, "round": round_num, "injuries": injuries, "wear_gain": wear_gain, "opponent_overall": opponent_overall } def generate_injuries(player: Fighter, won: bool, method: str, is_title: bool) -> List[Injury]: """Gera lesões realistas baseadas em dados médicos.""" new_injuries = [] base_mult = 1.0 if not won: base_mult = 2.1 if method == "KO/TKO": base_mult *= 1.8 if is_title: base_mult *= 1.3 for tipo, data in INJURY_TYPES.items(): chance = data["chance_base"] * base_mult # Wear & tear aumenta chance chance += player.wear_tear * 0.015 chance = min(0.75, chance) if random.random() < chance: sev = random.randint(1, 5) if tipo == "concussao" and method == "KO/TKO" and not won: sev = max(sev, 3) semanas = random.randint(*data["recuperacao"]) * (1 + sev // 3) permanente = False if tipo in ["concussao", "fratura"] and sev >= 4 and player.wear_tear > 25: permanente = random.random() < 0.25 new_injuries.append(Injury(tipo, sev, semanas, data["efeito"], permanente)) return new_injuries # ============================================================ # SISTEMA DE CARREIRA / RANKING # ============================================================ def get_opponent_overall(player: Fighter) -> int: stage = player.stage base = 40 + stage * 8 variance = random.randint(-12, 12) # No ranking, oponentes são mais fortes if player.ranking > 0: base += (16 - player.ranking) * 1.5 return max(35, min(95, int(base + variance))) def update_ranking(player: Fighter, result: dict): if player.stage < 2: # Ainda não no UFC return if result["won"]: if player.ranking == 0: # Entrou no ranking player.ranking = random.randint(12, 15) else: jump = 1 if result["method"] in ["KO/TKO", "Submission"]: jump += 1 if result["opponent_overall"] > player.overall + 5: jump += 1 player.ranking = max(1, player.ranking - jump) else: if player.ranking > 0: player.ranking = min(15, player.ranking + random.randint(2, 4)) if player.ranking > 15: player.ranking = 0 def check_stage_progression(player: Fighter): wins = player.record[0] if player.stage == 0 and wins >= random.randint(5, 8): player.stage = 1 print("\n★ Você está no radar das grandes promoções!") elif player.stage == 1 and wins >= random.randint(9, 14): player.stage = 2 player.fights_in_ufc = 0 print("\n★★★ CONTRATO UFC ASSINADO! ★★★") player.money += 25000 elif player.stage == 2 and player.ranking > 0: player.stage = 3 elif player.stage == 3 and player.ranking <= 5: player.stage = 4 print("\n★ Você entrou no Top 5! Title shot está próximo...") elif player.stage == 4 and player.ranking <= 2 and random.random() < 0.55: # Chance de title shot return True return False def offer_title_shot(player: Fighter) -> bool: print("\n" + "="*50) print("🏆 OPORTUNIDADE DE TITLE SHOT 🏆") print("="*50) print(f"Você está ranqueado #{player.ranking}.") print("A promoção oferece a chance pelo cinturão.") escolha = input("Aceitar a disputa de cinturão? (s/n): ").strip().lower() return escolha == "s" # ============================================================ # INTERFACE E LOOP PRINCIPAL # ============================================================ def clear(): os.system("cls" if os.name == "nt" else "clear") def print_header(player: Fighter): print("="*60) print(f" {player.nome} | {player.peso} | Idade: {player.idade}") print(f" Record: {player.record[0]}-{player.record[1]}-{player.record[2]} | Overall: {player.overall}") print(f" Stage: {STAGES[player.stage]} | Ranking: {'#'+str(player.ranking) if player.ranking else 'Unranked'}") print(f" Dinheiro: ${player.money:,} | Hype: {player.hype} | Wear&Tear: {player.wear_tear:.1f}") if player.active_injuries(): print(" LESÕES ATIVAS:") for inj in player.active_injuries(): perm = " (PERMANENTE)" if inj.permanente else "" print(f" - {inj.tipo.upper()} sev.{inj.severidade} | {inj.semanas_restantes} semanas{perm}") print("="*60) def train(player: Fighter): print("\n--- TREINO ---") print("1. Striking") print("2. Grappling") print("3. Cardio / Conditioning") print("4. Sparring (risco de lesão)") escolha = input("Escolha: ").strip() gain = random.randint(1, 3) + player.academy_level // 2 if escolha == "1": player.atributos["striking"] = min(99, player.atributos["striking"] + gain) player.atributos["power"] = min(99, player.atributos["power"] + gain // 2) print(f"Striking melhorou +{gain}") elif escolha == "2": player.atributos["grappling"] = min(99, player.atributos["grappling"] + gain) player.atributos["clinch"] = min(99, player.atributos["clinch"] + gain // 2) print(f"Grappling melhorou +{gain}") elif escolha == "3": player.atributos["cardio"] = min(99, player.atributos["cardio"] + gain) player.atributos["mental"] = min(99, player.atributos["mental"] + 1) print(f"Cardio melhorou +{gain}") elif escolha == "4": player.atributos["fight_iq"] = min(99, player.atributos["fight_iq"] + gain) if random.random() < 0.18: inj = Injury("lesao_muscular", random.randint(1, 3), random.randint(2, 5), "forca_reduzida") player.injuries.append(inj) print("⚠ Você se lesionou no sparring!") else: print(f"Sparring produtivo. (s/n): ").strip().lower() for inj in player.active_injuries(): if escolha == "s" and player.money >= custo: inj.semanas_restantes = max(0, inj.semanas_restantes - random.randint(2, 4)) player.money -= custo print(f"Tratamento aplicado em {inj.tipo}.") else: inj.semanas_restantes = max(0, inj.semanas_restantes - 1) print("Recuperação avançou.") def weight_cut(player: Fighter): print("\n--- WEIGHT CUT ---") print("1. Você está no peso com segurança.") def do_fight(player: Fighter, is_title: bool = False): if player.is_injured() and any(i.severidade >= 3 for i in player.active_injuries()): print("⚠ Você está seriamente lesionado. (s/n): ").strip().lower() if escolha != "s": return weight_cut(player) opp = get_opponent_overall(player) print(f"\nOponente Overall: {opp}") print("A luta vai começar...") time.sleep(1.2) result = simulate_fight(player, opp, is_title) if result["won"]: player.record[0] += 1 print(f"\n✅ VOCÊ VENCEU por {result['method']} (Round {result['round']})!") purse = 8000 + player.stage * 6000 + player.hype * 200 if is_title: purse *= 3 player.stage = 5 player.title_defenses = 0 print("\n🏆🏆🏆 VOCÊ É O NOVO CAMPEÃO DO UFC! 🏆🏆🏆") player.money += purse player.hype += random.randint(3, 8) else: player.record[1] += 1 print(f"\n❌ Você perdeu por {result['method']} (Round {result['round']}).") purse = 4000 + player.stage * 2500 player.money += purse player.hype = max(5, player.hype - random.randint(2, 6)) if is_title: player.stage = 4 player.ranking = random.randint(2, 5) print("Você perdeu o cinturão...") player.wear_tear += result["wear_gain"] player.injuries.extend(result["injuries"]) if result["injuries"]: print("\nLesões sofridas:") for inj in result["injuries"]: print(f" - {inj.tipo.upper()} (severidade {inj.severidade})") update_ranking(player, result) if player.stage >= 2: player.fights_in_ufc += 1 # Avanço de estágio title_offered = check_stage_progression(player) if title_offered and not is_title: if offer_title_shot(player): do_fight(player, is_title=True) return # Defesa de cinturão if player.stage == 5 and not is_title: player.title_defenses += 1 print(f"\nDefesas de cinturão: {player.title_defenses}") if random.random() < 0.4: print("Nova defesa de cinturão marcada!") do_fight(player, is_title=True) player.idade += 0 if random.random() < 0.7 else 1 # idade avança ocasionalmente time.sleep(2) def main_menu(player: Fighter): while not player.retired: clear() print_header(player) print("\nO que deseja fazer?") print("1.
